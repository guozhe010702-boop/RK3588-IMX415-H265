cmake_minimum_required(VERSION 3.16)
project(rk_yoloworld LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
include_directories(include)
find_package(Threads REQUIRED)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE Threads::Threads)
add_test(NAME core COMMAND core_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(capture_tests tests/capture_tests.cpp src/capture.cpp)
  target_link_options(capture_tests PRIVATE -Wl,--wrap=open,--wrap=close,--wrap=mmap,--wrap=munmap,--wrap=ioctl,--wrap=poll)
  add_test(NAME capture_lifecycle COMMAND capture_tests)
endif()
option(CORE_ONLY "Build portable queue/format tests only" OFF)
option(ADAPTER_TESTS "Test model adapter with explicit mock runtime" OFF)
if(ADAPTER_TESTS)
  add_library(mock_rknn SHARED tests/mock_rknn.cpp)
  target_include_directories(mock_rknn PRIVATE vendor/include)
  set_target_properties(mock_rknn PROPERTIES OUTPUT_NAME rknnrt)
  add_executable(model_tests tests/model_tests.cpp src/model.cpp vendor/tokenizer/clip_tokenizer.cpp)
  target_include_directories(model_tests PRIVATE vendor/include vendor/tokenizer)
  target_link_libraries(model_tests PRIVATE dl Threads::Threads)
  add_test(NAME model_adapter COMMAND model_tests)
  set_tests_properties(model_adapter PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_BINARY_DIR}")
endif()
if(CORE_ONLY)
  return()
endif()
set(MUXER_PREFIX "${CMAKE_SOURCE_DIR}/build/muxer" CACHE PATH "Private NUT-only FFmpeg libraries")
add_library(nut_output src/output.cpp)
target_include_directories(nut_output PRIVATE "${MUXER_PREFIX}/include")
target_link_libraries(nut_output PUBLIC "${MUXER_PREFIX}/lib/libavformat.so" "${MUXER_PREFIX}/lib/libavcodec.so" "${MUXER_PREFIX}/lib/libavutil.so" Threads::Threads)
add_executable(nut_fixture tests/nut_fixture.cpp)
target_link_libraries(nut_fixture PRIVATE nut_output)
add_executable(output_failure_tests tests/output_failure_tests.cpp)
target_link_libraries(output_failure_tests PRIVATE nut_output)
add_executable(picture_parity tests/picture_parity.cpp src/model.cpp vendor/tokenizer/clip_tokenizer.cpp)
target_include_directories(picture_parity PRIVATE vendor/include vendor/tokenizer)
target_link_libraries(picture_parity PRIVATE dl Threads::Threads)
add_executable(io_smoke tests/io_smoke.cpp src/capture.cpp src/preprocess.cpp)
target_include_directories(io_smoke PRIVATE vendor/include/rga)
target_link_libraries(io_smoke PRIVATE nut_output "${CMAKE_SOURCE_DIR}/vendor/librga.so")
add_executable(rknn_yolo_world_realtime src/main.cpp src/capture.cpp src/model.cpp src/preprocess.cpp src/draw.cpp vendor/tokenizer/clip_tokenizer.cpp)
target_include_directories(rknn_yolo_world_realtime PRIVATE vendor/include vendor/include/rga vendor/tokenizer)
target_link_libraries(rknn_yolo_world_realtime PRIVATE nut_output "${CMAKE_SOURCE_DIR}/vendor/librga.so" dl)
set_target_properties(rknn_yolo_world_realtime nut_fixture PROPERTIES BUILD_RPATH "${MUXER_PREFIX}/lib" INSTALL_RPATH "$ORIGIN/lib")
